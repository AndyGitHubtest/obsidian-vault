### Step 2: 低波动压缩识别（蓄力检测）
```python
def detect_volatility_compression(bars, lookback=12):
    """
    检测波动率压缩 - 弹簧被压紧
    条件: 连续N根K线波动率低于阈值
    """
    recent = bars[-lookback:]
    
    # ATR 压缩
    atr_values = [calc_atr(recent[i:i+14]) for i in range(len(recent)-13)]
    avg_atr_pct = np.mean(atr_values) / bars[-1].close
    
    # 布林带收窄
    bb_width = (bollinger_upper - bollinger_lower) / bollinger_mid
    
    # K线实体变小
    body_sizes = [abs(b.close - b.open) / b.open for b in recent]
    avg_body = np.mean(body_sizes)
    
    # 压缩信号
    compression = (
        avg_atr_pct < 0.008 and      # ATR < 0.8%
        bb_width < 0.02 and           # 带宽 < 2%
        avg_body < 0.003              # 实体 < 0.3%
    )
    
    return compression, avg_atr_pct
```

def detect_volume_surge(bar, vol_ma, threshold=2.5):

### Step 4: 价格突破确认（触发信号）
```python
def detect_price_breakout(bar, high_N, breakout_threshold=0.005):
    """
    价格突破 N 周期高点
    """
    # 收盘价突破
    price_break = bar.close > high_N
    
    # 突破幅度（避免假突破）
    breakout_strength = (bar.close - high_N) / high_N
    valid_breakout = breakout_strength > breakout_threshold
    
    # 趋势方向确认
    trend_ok = bar.ema_fast > bar.ema_slow
    
    return price_break and valid_breakout and trend_ok
```

### Step 5: 动量强度评分（质量筛选）
```python
def calc_momentum_score(bar, prev_bars):
    """
    动量强度评分 (0-100分)
    """
    score = 0
    
    # 1. 突破强度 (0-25分)
    breakout_pct = (bar.close - prev_bars.resistance) / prev_bars.resistance
    score += min(breakout_pct / 0.02, 1.0) * 25
    
    # 2. 成交量强度 (0-25分)
    vol_ratio = bar.volume / prev_bars.vol_ma
    score += min(vol_ratio / 3.0, 1.0) * 25
    
    # 3. 波动率强度 (0-25分)
    atr_expansion = bar.atr / prev_bars.atr_ma
    score += min(atr_expansion / 2.0, 1.0) * 25
    
    # 4. 趋势强度 (0-25分)
    ema_spread = (bar.ema_fast - bar.ema_slow) / bar.ema_slow
    score += min(ema_spread / 0.01, 1.0) * 25
    
    return score

# 评分阈值
MOMENTUM_THRESHOLD = 60  # ≥60分才入场
```

### Step 6: 延续概率评估（胜率预测）
```python
def calc_continuation_probability(bar, market_context):
    """
    评估趋势延续概率 (0-100%)
    """
    prob = 0
    
    # 趋势强度 (30%权重)
    adx = calc_adx(bar, 14)
    if adx > 25:
        prob += 30
    elif adx > 20:
        prob += 15
    
    # 成交量配合 (25%权重)
    vol_ratio = bar.volume / bar.vol_ma
    if vol_ratio > 2.5:
        prob += 25
    elif vol_ratio > 2.0:
        prob += 15
    
    # 波动率状态 (20%权重) - 从压缩到扩张
    if bar.atr_expansion > 1.5:
        prob += 20
    elif bar.atr_expansion > 1.2:
        prob += 10
    
    # 市场环境 (15%权重)
    if market_context.btc_trend == 'bullish':
        prob += 15
    elif market_context.btc_trend == 'neutral':
        prob += 8
    
    # 时间因子 (10%权重)
    if is_active_trading_hours(bar.time):
        prob += 10
    
    return prob

# 延续概率阈值
CONTINUATION_THRESHOLD = 55  # ≥55%才入场
```

## 四、进场逻辑

### 入场条件汇总
```python
def check_entry(bars, bar, market_context):
    """
    完整入场检查 - 10步漏斗
    """
    # Step 1: 全市场扫描 (已在外部完成)
    
    # Step 2: 低波动压缩
    compression, avg_atr = detect_volatility_compression(bars)
    if not compression:
        return False, '无波动率压缩'
    
    # Step 3: 异常放量
    vol_surge, vol_ratio = detect_volume_surge(bar, bars.vol_ma)
    if not vol_surge:
        return False, '无量能放大'
    
    # Step 4: 价格突破
    breakout = detect_price_breakout(bar, bars.high_N)
    if not breakout:
        return False, '无价格突破'
    
    # Step 5: 动量评分
    score = calc_momentum_score(bar, bars)
    if score < MOMENTUM_THRESHOLD:
        return False, f'动量评分不足: {score}'
    
    # Step 6: 延续概率
    prob = calc_continuation_probability(bar, market_context)
    if prob < CONTINUATION_THRESHOLD:
        return False, f'延续概率不足: {prob}%'
    
    return True, f'信号通过: score={score}, prob={prob}%'
```

### 分批建仓（加仓逻辑）
```python
class BatchEntry:
    """
    分批建仓
    逻辑: 信号确认后分3次进场，每次确认趋势延续再加仓
    """
    def __init__(self, signal, total_position_size):
        self.signal = signal
        self.total_size = total_position_size
        self.batches = [
            {'ratio': 0.4, 'filled': False},  # 首仓 40%
            {'ratio': 0.3, 'filled': False},  # 加仓 30%
            {'ratio': 0.3, 'filled': False},  # 加仓 30%
        ]
        self.current_batch = 0
        self.filled_size = 0
    
    def check_add_position(self, bar, highest_price):
        """检查是否可以加仓"""
        if self.current_batch >= 2:  # 首仓已进，最多加2次
            return None
        
        next_batch = self.current_batch + 1
        
        # 加仓条件: 价格继续向有利方向移动
        profit_ratio = (highest_price - self.signal.entry_price) / self.signal.risk
        
        if next_batch == 1 and profit_ratio >= 0.5:  # 盈利 0.5R 加第一批
            add_qty = self.total_size * self.batches[next_batch]['ratio']
            self.current_batch = next_batch
            self.filled_size += add_qty
            self.batches[next_batch]['filled'] = True
            return add_qty
        
        if next_batch == 2 and profit_ratio >= 1.0:  # 盈利 1.0R 加第二批
            add_qty = self.total_size * self.batches[next_batch]['ratio']
            self.current_batch = next_batch
            self.filled_size += add_qty
            self.batches[next_batch]['filled'] = True
            return add_qty
        
        return None
```

## 五、出场逻辑（核心）

### 分批止盈
```python
class BatchExit:
    """分批止盈管理"""
    
    def __init__(self, entry_price, stop_loss, position_size):
        self.entry_price = entry_price
        self.stop_loss = stop_loss
        self.risk = entry_price - stop_loss  # 单笔风险
        self.position_size = position_size
        self.remaining = position_size
        
        # 分批止盈目标
        self.targets = [
            {'ratio': 0.4, 'price': entry_price + self.risk * 2},  # 1:2
            {'ratio': 0.3, 'price': entry_price + self.risk * 3},  # 1:3
            {'ratio': 0.3, 'price': None},  # 移动止盈
        ]
        self.batch_idx = 0
    
    def check_exit(self, current_price, highest_price):
        """检查是否触发分批止盈"""
        exits = []
        
        if self.batch_idx < 2:  # 前两批固定目标
            target = self.targets[self.batch_idx]
            if current_price >= target['price']:
                exit_qty = self.position_size * target['ratio']
                exits.append({'price': target['price'], 'qty': exit_qty})
                self.remaining -= exit_qty
                self.batch_idx += 1
        
        # 第三批移动止盈
        if self.batch_idx == 2 and self.remaining > 0:
            trailing_stop = highest_price - ATR * 2.0
            if current_price <= trailing_stop:
                exits.append({'price': current_price, 'qty': self.remaining})
                self.remaining = 0
        
        return exits
```

### 移动止盈
```python
class TrailingStop:
    """移动止盈"""
    
    def __init__(self, entry_price, atr, trailing_multiplier=2.0):
        self.entry_price = entry_price
        self.atr = atr
        self.multiplier = trailing_multiplier
        self.highest_price = entry_price
        self.trailing_line = entry_price - atr * trailing_multiplier
        self.activated = False  # 是否已激活
    
    def update(self, current_price, current_atr):
        """更新移动止损线"""
        self.highest_price = max(self.highest_price, current_price)
        
        # 盈利超过 1:2 后激活
        profit_ratio = (current_price - self.entry_price) / (self.entry_price - self.stop_loss)
        if profit_ratio >= 2.0:
            self.activated = True
        
        if self.activated:
            new_line = self.highest_price - current_atr * self.multiplier
            self.trailing_line = max(self.trailing_line, new_line)
        
        return self.trailing_line
    
    def is_triggered(self, current_price):
        """检查是否触发移动止损"""
        return current_price <= self.trailing_line
```

### Step 8: 失效检测退出
```python
class InvalidationDetector:
    """
    失效检测 - 当突破信号被证伪时立即退出
    """
    def __init__(self, entry_price, breakout_level, atr):
        self.entry_price = entry_price
        self.breakout_level = breakout_level  # 突破的关键位
        self.atr = atr
        self.highest_price = entry_price
        self.time_in_trade = 0  # 持仓K线数
    
    def check_invalidated(self, bar):
        """
        检查信号是否失效
        """
        reasons = []
        
        # 1. 价格跌回突破位下方 (假突破)
        if bar.close < self.breakout_level:
            reasons.append('跌回突破位，假突破')
        
        # 2. 持仓时间过长无盈利 (动能衰竭)
        self.time_in_trade += 1
        if self.time_in_trade > 24:  # 2小时无进展
            if (bar.close - self.entry_price) / self.entry_price < 0.005:
                reasons.append('持仓2小时无盈利，动能衰竭')
        
        # 3. 成交量急剧萎缩 (参与者撤离)
        if bar.volume < bar.vol_ma * 0.5:
            reasons.append('成交量萎缩至均量50%以下')
        
        # 4. 反向动量出现 (反方向突破)
        if bar.close < bar.ema_fast and bar.close < bar.ema_slow:
            reasons.append('EMA 死叉，趋势反转')
        
        # 5. 波动率异常收缩 (行情结束)
        if bar.atr < self.atr * 0.5:
            reasons.append('波动率收缩50%，行情结束')
        
        if reasons:
            return True, ', '.join(reasons)
        
        return False, None
```

## 六、风控模块

### 风控检查
```python
class RiskManager:
    def __init__(self, capital, config):
        self.capital = capital
        self.max_position_pct = config['max_position_pct']  # 5%
        self.max_positions = config['max_positions']  # 3
        self.daily_loss_limit = config['daily_loss_limit']  # 10%
        self.current_positions = {}
        self.daily_pnl = 0
    
    def check_entry(self, signal, position_size):
        """检查是否允许进场"""
        # 1. 仓位上限检查
        position_value = position_size * signal.price
        if position_value > self.capital * self.max_position_pct:
            return False, '仓位超限'
        
        # 2. 持仓数量检查
        if len(self.current_positions) >= self.max_positions:
            return False, '持仓数已达上限'
        
        # 3. 日止损检查
        if self.daily_pnl < -self.capital * self.daily_loss_limit:
            return False, '日止损已触发'
        
        # 4. 互斥检查
        if signal.symbol in self.current_positions:
            return False, '该品种已有持仓'
        
        return True, '通过'
```

## 七、异常处理

| 场景 | 处理策略 |
|------|----------|
| API 超时 | 重试 3 次，失败报警 |
