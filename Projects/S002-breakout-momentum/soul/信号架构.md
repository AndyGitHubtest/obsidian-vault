```python
def filter_symbols(symbols):
    """全市场标的过滤"""
    filtered = []
    for s in symbols:
        # 流动性过滤
        if s.daily_volume < 2_000_000:  # 200万U
            continue
        
        # 数据质量过滤
        if s.missing_rate > 0.05:  # 缺失率>5%
            continue
        
        # 垃圾币过滤
        if is_meme(s) or is_single_char(s) or is_stablecoin(s):
            continue
        
        # 盘口质量过滤
        if s.spread_pct > 0.005:  # 买卖价差>0.5%
            continue
        if s.depth_1pct < 50_000:  # 1%盘口深度<5万U
            continue
        
        # 异常数据过滤
        if s.has_extreme_spikes:  # 频繁插针
            continue
        if s.avg_slippage > market_avg * 2:  # 滑点过高
            continue
        
        filtered.append(s)
    return filtered
```

### 4.5 K线实体压缩
```python
def detect_body_compression(bars, period=12):
    """K线实体压缩: 平均实体长度下降"""
    bodies = [abs(b.close - b.open) / b.open for b in bars[-period:]]
    avg_body = np.mean(bodies)
    upper_shadows = [(b.high - max(b.open, b.close)) / b.open for b in bars[-period:]]
    lower_shadows = [(min(b.open, b.close) - b.low) / b.open for b in bars[-period:]]
    
    # 实体缩小 + 影线变短 + 连续小实体增多
    small_bodies = sum(1 for b in bodies if b < 0.003)
    
    compression = (
        avg_body < 0.003 and
        small_bodies > period * 0.6  # 60% 以上是小实体
    )
    return compression, avg_body
```

### 4.6 压缩质量过滤
```python
def check_compression_quality(bars, period=24):
    """
    压缩期间质量检查
    - 缩量但不死（成交量不能归零）
    - 价格结构健康（箱体/旗形/三角/平台）
    - 无异常插针
    """
    # 成交量检查：缩量但不死
    vol_recent = np.mean(bars.volume[-period:])
    vol_past = np.mean(bars.volume[-period*2:-period])
    vol_ratio = vol_recent / vol_past
    
    if vol_ratio < 0.2:  # 成交量萎缩到 20% 以下 = 没人玩
        return False, '死币横盘'
    
    # 价格结构检查：不能持续阴跌
    recent_returns = bars.returns[-period:]
    if sum(1 for r in recent_returns if r < 0) > period * 0.7:
        return False, '持续阴跌'
    
    # 异常插针过滤
    max_spikes = sum(1 for b in bars[-period:] 
                     if (b.high - b.low) / b.open > 0.03)  # 单根振幅>3%
    if max_spikes > 3:
        return False, '频繁插针'
    
    # 结构判断：箱体/旗形/三角/平台
    structure = classify_structure(bars[-period:])
    
    return True, f'结构健康: {structure}'
```

## 五、层 3 - 爆发确认（触发阶段）

### 5.1 价格突破确认
```python
def detect_price_breakout(bars, period=20):
    """
    收盘价突破近 N 根高点
    优先看收盘突破，盘中刺破容易假突破
    """
    high_N = np.max(bars.high[-period-1:-1])  # 不包含当前K线
    current_close = bars.close[-1]
    
    # 收盘突破
    breakout = current_close > high_N
    
    # 突破幅度
    breakout_pct = (current_close - high_N) / high_N
    
    # 突破实体质量
    body_size = abs(bars.close[-1] - bars.open[-1]) / bars.open[-1]
    upper_shadow = (bars.high[-1] - max(bars.open[-1], bars.close[-1])) / bars.open[-1]
    close_position = (bars.close[-1] - bars.low[-1]) / (bars.high[-1] - bars.low[-1])
    
    # 质量检查
    quality = (
        breakout_pct > 0.005 and      # 突破幅度>0.5%
        body_size > 0.005 and          # 实体>0.5%
        upper_shadow < body_size * 0.5 and  # 上影不超过实体一半
        close_position > 0.7           # 收盘位置在K线上70%
    )
    
    return breakout and quality, breakout_pct
```

### 5.2 收益率异常确认
```python
def detect_return_anomaly(bars, period=20):
    """
    收益率 Z-score: ret_z = (ret_t - mean) / std
    统计意义上的"异常动量"
    """
    returns = bars.returns[-period:]
    ret_t = bars.returns[-1]
    ret_mean = np.mean(returns[:-1])
    ret_std = np.std(returns[:-1])
    
    if ret_std == 0:
        return False, 0
    
    z_score = (ret_t - ret_mean) / ret_std
    return z_score > 2.0, z_score  # 超过2个标准差
```

### 5.3 连续位移确认（P2 修复：OR 逻辑替代 AND）
```python
def detect_continuous_move(bars):
    """
    不是单根暴涨，而是短时间连续上涨
    P2 修复: 三个条件同时满足过于严格，改为 OR 逻辑
    """
    cum_ret_3 = sum(bars.returns[-3:])
    cum_ret_5 = sum(bars.returns[-5:])
    cum_ret_8 = sum(bars.returns[-8:])
    
    # P2 修复: OR 逻辑，满足任一即可
    # 但给不同权重
    score = 0
    if cum_ret_3 > 0.015:  # 3根累计>1.5%
        score += 1
    if cum_ret_5 > 0.025:  # 5根累计>2.5%
        score += 1
    if cum_ret_8 > 0.035:  # 8根累计>3.5%
        score += 1
    
    # 至少满足 2 个条件（OR 但有最低要求）
    confirmed = score >= 2
    
    return confirmed, cum_ret_5, score
```

### 5.4 成交量确认
```python
def detect_volume_confirmation(bars, period=20):
    """
    成交量确认: 单根异常 + 连续放量 + 价量同步
    """
    vol_ma = np.mean(bars.volume[-period:])
    
    # 单根量比
    vol_ratio_now = bars.volume[-1] / vol_ma
    
    # 连续放量（2-3根累计量）
    vol_cum_2 = sum(bars.volume[-2:]) / (vol_ma * 2)
    vol_cum_3 = sum(bars.volume[-3:]) / (vol_ma * 3)
    
    # 价量同步
    price_up = bars.close[-1] > bars.open[-1]
    vol_up = bars.volume[-1] > vol_ma
    vol_sync = price_up and vol_up
    
    # 量价背离检测
    if bars.close[-1] > bars.open[-1] and bars.volume[-1] < vol_ma * 0.8:
        vol_divergence = True  # 价涨量缩，降权
    else:
        vol_divergence = False
    
    confirmed = (
        vol_ratio_now > 2.0 and
        vol_cum_3 > 1.5 and
        vol_sync and
        not vol_divergence
    )
    
    return confirmed, vol_ratio_now
```

```python
def calc_entry_score(bars, symbol, market_context):
    """
    入场评分 = 压缩分(30%) + 爆发分(45%) + 执行分(25%)
    0-100 分，仅用突破前/突破时可见数据
    
    P0 修复说明:
    - 移除延续评分（回撤/站稳/量能维持是事后数据，实盘不可见）
    - 延续判断改为持仓期间的动态评估（见持仓管理模块）
    """
    compression_score = calc_compression_score(bars)     # 0-100
    breakout_score = calc_breakout_score(bars)           # 0-100
    execution_score = calc_execution_score(symbol)       # 0-100
    
    total = (
        compression_score * 0.30 +
        breakout_score * 0.45 +
        execution_score * 0.25
    )
    
    return total, {
        'compression': compression_score,
        'breakout': breakout_score,
        'execution': execution_score
    }
```

```python
def calc_compression_score(bars):
    """压缩评分: 压得越紧分越高，全部用突破前数据"""
    score = 0
    
    # 波动率压缩程度 (25分)
    vol_ratio = calc_vol_ratio(bars)
    score += max(0, 25 - vol_ratio * 50)  # ratio越低分越高
    
    # ATR压缩程度 (25分)
    atr_ratio = calc_atr_ratio(bars)
    score += max(0, 25 - atr_ratio * 40)
    
    # 布林带宽压缩 (25分)
    bb_percentile = calc_bb_percentile(bars)
    score += max(0, 25 - bb_percentile * 25)
    
    # 区间压缩程度 (25分)
    range_ratio = calc_range_ratio(bars)
    score += max(0, 25 - range_ratio * 500)
    
    return max(0, min(100, score))
```

```python
def calc_execution_score(symbol):
    """执行评分: 可交易性评估，静态数据"""
    score = 0
    
    # 滑点大小 (35分)
    slippage = get_avg_slippage(symbol)
    score += max(0, 35 - slippage * 1000)
    
    # 盘口深度 (35分)
    depth = get_depth(symbol)

    return max(0, min(100, score))
```

## 五、持仓管理 - 延续动态评估（P0 修复：延续判断移到持仓阶段）

```python
class PositionContinuationTracker:
    """
    P0 修复: 延续判断不在入场时做，而是在持仓期间动态评估
    持仓期间这些数据才实时存在
    """
    def __init__(self, position, bars_history):
        self.position = position
        self.highest = position.entry_price
        self.bars_since_entry = 0
        self.max_profit = 0
        self.pullback_from_high = 0
        self.bars_above_breakout = 0
        self.total_bars_above = 0
    
    def update(self, bar):
        """每根K线更新延续状态"""
        self.bars_since_entry += 1
        self.highest = max(self.highest, bar.high)
        
        # 回撤深度
        current_profit = (bar.close - self.position.entry_price) / self.position.risk
        self.max_profit = max(self.max_profit, current_profit)
        self.pullback_from_high = self.max_profit - current_profit
        
        # 突破位站稳
        if bar.close > self.position.breakout_level:
            self.bars_above_breakout += 1
        self.total_bars_above = self.bars_above_breakout
    
    def calc_continuation_health(self):
        """
        持仓健康度评估（0-100）
        用于决定: 是否加仓 / 是否提前减仓 / 是否收紧止损
        """
        score = 50  # 基础分
        
        # 回撤浅度 (25分)
        if self.pullback_from_high < 0.2:
            score += 25
        elif self.pullback_from_high < 0.5:
            score += 15
        elif self.pullback_from_high > 1.0:
            score -= 25
        
        # 价格效率 (25分)
        if self.bars_since_entry > 0:
            efficiency = self.max_profit / (self.bars_since_entry * 0.01)  # 归一化
            score += min(efficiency, 1.0) * 25
        
        # 突破位站稳比例 (25分)
        if self.bars_since_entry > 0:
            stand_ratio = self.bars_above_breakout / self.bars_since_entry
            score += stand_ratio * 25
        
        # 持续盈利时间 (25分)
        if self.max_profit > 0.5:
            score += 25
        elif self.max_profit > 0:
            score += 10
        
        return max(0, min(100, score))
    
    def get_action_recommendation(self):
        """根据延续健康度推荐操作"""
        health = self.calc_continuation_health()
        
        if health >= 75:
            return 'HOLD_STRONG', '趋势强劲，继续持有或考虑加仓'
        elif health >= 50:
            return 'HOLD', '趋势正常，继续持有'
        elif health >= 30:
            return 'TIGHTEN_STOP', '趋势减弱，收紧止损'
        else:
            return 'EXIT', '趋势破坏，建议退出'
```

### 7.1 入场逻辑（用户修正：必须回调进场，不追高）
```python
def check_entry(bars, score, market_context, all_candidate_scores, breakout_signal):
    """
    入场原则:
    - 必须回调进场，绝不在突破那根K线追高
    - 等待回踩确认 + 不破关键支撑 + 再次启动
    
    入场流程:
    1. 突破信号已触发（记录突破位、突破K线）
    2. 等待回踩：价格回到突破位/EMA/前高附近
    3. 回踩不破：关键支撑位没有被有效跌破
    4. 再次启动：回踩后出现企稳K线（小阳/十字星/放量阳线）
    5. 评分达标：Top N 排名 + 最低质量线
    """
    # 最低质量线
    MIN_QUALITY_SCORE = 40
    
    if score < MIN_QUALITY_SCORE:
        return False, f'信号质量过低: {score}'
    
    # 检查是否满足回调进场条件
    pullback_ok, pullback_reason = check_pullback_entry(bars, breakout_signal)
    if not pullback_ok:
        return False, pullback_reason  # '未回踩' / '回踩过深' / '未企稳'
    
    # Top N 排名
    TOP_N = 5
    rank = calc_signal_rank(score, all_candidate_scores)
    
    if rank > TOP_N:
        return False, f'信号排名靠后: #{rank}'
    
    # 避免量能衰减
    vol_recent = np.mean(bars.volume[-2:])
    vol_past = np.mean(bars.volume[-5:-2])
    if vol_recent < vol_past * 0.7:
        return False, '量能衰减'
    
    # 市场环境过滤
    if market_context.env == '冷清死水':
        return False, '死水市场'
    if market_context.env == '混沌震荡':
        score *= 0.5  # 降权
    
    return True, f'回调进场: score={score}, rank=#{rank}'


def check_pullback_entry(bars, breakout_signal):
    """
    检查是否满足回调进场条件
    
    回调进场 4 个条件:
    1. 已经发生过回踩（价格回到突破位附近）
    2. 回踩没有跌破关键支撑（突破位/EMA9/前低）
    3. 回踩后出现企稳信号（小实体K线/缩量）
    4. 再次启动信号（放量阳线/突破回踩高点）
    """
    entry_price = breakout_signal.entry_price
    breakout_level = breakout_signal.breakout_level
    hold_bars = breakout_signal.hold_bars
    
    # 条件 1: 至少等待 2 根K线（给回踩留出时间）
    if hold_bars < 2:
        return False, '突破刚发生，等待回踩'
    
    # 条件 2: 检查是否发生过回踩
    # 回踩定义：最低价回到突破位附近（±0.5%）
    pullback_tolerance = 0.005  # 0.5%
    pullback_depth = (breakout_level - min(bars.low[-hold_bars:])) / breakout_level
    
    # 回踩过深（跌破突破位太多 = 假突破概率高）
    if pullback_depth > 0.02:  # 跌破超过 2%
        return False, f'回踩过深: {pullback_depth:.1%}'
    
    # 没有回踩（价格一直在突破位上方远离）
    min_distance = min((bar.low - breakout_level) / breakout_level for bar in bars[-hold_bars:])
    if min_distance > 0.01:  # 最低价都离突破位>1% = 没回踩
        return False, '尚未回踩，不追'
    
    # 条件 3: 回踩后企稳信号
    # 最近 1-2 根K线实体变小，成交量萎缩
    recent_bodies = [abs(b.close - b.open) / b.open for b in bars[-2:]]
    avg_recent_body = np.mean(recent_bodies)
    breakout_body = abs(breakout_signal.breakout_bar.close - breakout_signal.breakout_bar.open) / breakout_signal.breakout_bar.open
    
    if avg_recent_body > breakout_body * 0.8:
        return False, '回踩未企稳，实体仍然太大'
    
    # 条件 4: 再次启动信号（当前K线放量阳线）
    current_bar = bars[-1]
    if current_bar.close > current_bar.open:  # 阳线
        if current_bar.volume > np.mean(bars.volume[-3:-1]) * 1.2:  # 放量
            return True, '回踩企稳，再次启动'
    
    return False, '等待再次启动信号'
```

### 7.2 仓位管理
```python
class PositionManager:
    def __init__(self, capital, config):
        self.capital = capital
        self.base_risk_pct = config['base_risk_pct']  # 2%
        self.max_position_pct = config['max_position_pct']  # 5%
        self.max_positions = config['max_positions']  # 3
        self.consecutive_losses = 0
        self.env_multiplier = 1.0
    
    def calc_position_size(self, signal_score, entry_price, stop_price, market_env):
        """
        仓位计算: 分数加权 + 环境调仓 + 连续亏损降档
        """
        # 基础风险金额
        risk_amount = self.capital * self.base_risk_pct
        
        # 分数加权 (分数越高仓位越大)
        score_multiplier = 0.5 + (signal_score / 100) * 1.0  # 0.5x ~ 1.5x
        
        # 环境调仓
        env_mult = {
            '强趋势': 1.2,
            '活跃轮动': 1.0,
            '混沌震荡': 0.5,
            '冷清死水': 0.0
        }.get(market_env, 1.0)
        
        # 连续亏损降档
        loss_mult = max(0.3, 1.0 - self.consecutive_losses * 0.15)
        
        # 综合调整
        adjusted_risk = risk_amount * score_multiplier * env_mult * loss_mult
        
        # 止损距离反推仓位
        stop_distance = entry_price - stop_price
        position_size = adjusted_risk / stop_distance
        
        # 仓位上限检查
        max_size = (self.capital * self.max_position_pct) / entry_price
        position_size = min(position_size, max_size)
        
        return position_size
```

```python
def check_stop_loss(position, bar):
    """
    止损检查（多重保护）
    P1 修复: 时间止损改为动态调整
    """
    stops = []
    
    # 1. 跌回突破位下方
    if bar.close < position.breakout_level:
        stops.append(('突破位止损', bar.close))
    
    # 2. ATR 止损
    atr_stop = position.entry_price - position.atr * 1.5
    if bar.close <= atr_stop:
        stops.append(('ATR止损', atr_stop))
    
    # 3. 时间止损（P1 修复：动态调整）
    # 基础时间：24 根（2小时）
    # 如果已盈利，延长到 48 根（4小时）
    # 如果波动率高，延长到 36 根
    base_time_bars = 24
    if position.current_profit > 0.5:  # 盈利超过 0.5R
        time_limit = base_time_bars * 2  # 延长到 4 小时
    elif position.atr_expansion > 1.5:  # 波动率扩张
        time_limit = int(base_time_bars * 1.5)  # 延长到 3 小时
    else:
        time_limit = base_time_bars
    
    if position.hold_bars > time_limit:
        if position.current_profit < 0.3:  # 盈利不足 0.3R
            stops.append(('时间止损-无进展', bar.close))
    
    # 4. 前低止损
    if bar.close < position.recent_low:
        stops.append(('前低止损', bar.close))
    
    if stops:
        return True, stops[0]  # 返回最早触发的止损
    
    return False, None
```

## 九、止盈逻辑（分批+移动）

### 9.1 分批止盈结构（用户修正：首批覆盖止损 + 多周期压力位止盈）
```python
class BatchTakeProfit:
    """
    分批止盈: 4阶段
    核心原则:
    1. 第一次止盈必须覆盖止损（盈亏比 ≥ 1:1，确保不亏）
    2. 每次止盈基于不同时间周期的压力位，不是固定盈亏比
    3. 最后必须保留移动止盈（不可被固定目标替代，确保吃到大趋势尾部）
    
    止盈结构:
    第1批(40%): 5m 级别压力位 / 1:1 盈亏比（取先触发者）→ 确保覆盖止损
    第2批(25%): 15m 级别压力位 → 主利润段
    第3批(20%): 1h 级别压力位 → 大波段
    第4批(15%): 移动止盈（ATR 追踪，强制保留）→ 让利润奔跑，不可提前结束
    """
    def __init__(self, entry_price, stop_loss, atr, total_size, resistance_levels):
        self.entry = entry_price
        self.risk = entry_price - stop_loss  # 单笔风险
        self.atr = atr
        self.total_size = total_size
        self.remaining = total_size
        self.highest = entry_price
        
        # 计算各批次止盈价格（基于多周期压力位）
        r5m = resistance_levels.get('5m', entry_price + self.risk * 1.5)
        r15m = resistance_levels.get('15m', entry_price + self.risk * 2.5)
        r1h = resistance_levels.get('1h', entry_price + self.risk * 3.5)
        
        # 第一批止盈：取 5m 压力位 和 1:1 盈亏比 的较小值（确保至少覆盖止损）
        first_tp = min(r5m, entry_price + self.risk * 1.0)
        
        # 4阶段止盈（基于多周期压力位，最后必须保留移动止盈）
        self.phases = [
            {'ratio': 0.40, 'trigger': 'resistance_5m', 'target': first_tp, 
             'min_rr': 1.0, 'done': False},  # 5m压力位，至少1:1（覆盖止损）
            {'ratio': 0.25, 'trigger': 'resistance_15m', 'target': r15m, 
             'min_rr': 1.5, 'done': False},  # 15m压力位
            {'ratio': 0.20, 'trigger': 'resistance_1h', 'target': r1h, 
             'min_rr': 2.0, 'done': False},  # 1h压力位
            # 最后一部分永远交给移动止盈，确保大行情能吃到尾部
            {'ratio': 0.15, 'trigger': 'trailing_stop', 'target': None, 
             'min_rr': None, 'done': False, 'mandatory': True},  
        ]
    
    def update(self, bar):
        """更新止盈状态"""
        self.highest = max(self.highest, bar.high)
        exits = []
        
        for i, phase in enumerate(self.phases):
            if phase['done']:
                continue
            
            triggered = False
            exit_price = bar.close
            
            if phase['trigger'] == 'trailing_stop':
                # 移动止盈：最高价 - ATR×2.0
                trailing = self.highest - self.atr * 2.0
                if bar.close <= trailing:
                    triggered = True
                    exit_price = bar.close
            else:
                # 压力位止盈：达到目标价且盈亏比≥最小要求
                target = phase['target']
                min_rr = phase['min_rr']
                
                if target and bar.close >= target:
                    # 验证盈亏比
                    actual_rr = (bar.close - self.entry) / self.risk
                    if actual_rr >= min_rr:
                        triggered = True
                        exit_price = bar.close
            
            if triggered:
                exit_qty = self.total_size * phase['ratio']
                exits.append({
                    'phase': i + 1,
                    'price': exit_price,
                    'qty': exit_qty,
                    'reason': phase['trigger'],
                    'rr': (exit_price - self.entry) / self.risk
                })
                self.remaining -= exit_qty
                phase['done'] = True
        
        return exits
```

```python
class InvalidationDetector:
    """
    失效检测 - 当信号逻辑被破坏时立即退出
    """
    def __init__(self, entry_price, breakout_level, atr):
        self.entry = entry_price
        self.breakout_level = breakout_level
        self.atr = atr
        self.hold_bars = 0
    
    def check(self, bar):
        """5大失效条件"""
        reasons = []
        self.hold_bars += 1
        
        # 1. 放量突破后迅速跌回箱体（假突破）
        if bar.close < self.breakout_level:
            reasons.append('跌回突破位，假突破')
        
        # 2. 后续量能明显衰减
        if bar.volume < bar.vol_ma * 0.5:
            reasons.append('量能衰减至50%以下')
        
        # 3. 突破位站不稳
        if self.hold_bars > 6:
            close_above = sum(1 for b in bar.recent(6) if b.close > self.breakout_level)
            if close_above < 3:  # 6根中少于3根收在突破位上方
                reasons.append('突破位站不稳')
        
        # 4. 连续上影 / 大阴吞没
        if bar.close < bar.open and bar.close < bar.recent(1).low:
            reasons.append('大阴吞没突破K线')
        
        # 5. 市场整体风险转坏
        if bar.market_env == 'risk_off':
            reasons.append('市场环境恶化')
        
        if reasons:
            return True, ', '.join(reasons)
        
        return False, None
```
