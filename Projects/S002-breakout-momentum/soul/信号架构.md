### Step 2: 爆点条件检查
```python
def detect_breakout(bar, prev_bars):
    # 1. 价格突破 N 日高点
    price_break = bar.close > prev_bars.high_N
    
    # 2. 成交量放大 (量比 > 2)
    vol_ratio = bar.volume / prev_bars.vol_ma
    vol_surge = vol_ratio > 2.0
    
    # 3. 波动率突增 (ATR > 1.5倍均值)
    atr_ratio = bar.atr / prev_bars.atr_ma
    vol_expansion = atr_ratio > 1.5
    
    # 4. 趋势确认 (快线 > 慢线)
    trend_ok = bar.ema_fast > bar.ema_slow
    
    return price_break and vol_surge and vol_expansion and trend_ok
```

### Step 3: 过滤条件
```python
def filter_signal(bar):
    # RSI 不在极端区域
    if bar.rsi > 80 or bar.rsi < 20:
        return False
    
    # 波动率不能太低（排除震荡市）
    if bar.atr / bar.close < 0.005:
        return False
    
    # 资金费率正常
    if abs(bar.funding_rate) > 0.001:
        return False
    
    return True
```

## 四、进场逻辑

### 仓位计算
```python
def calc_position_size(capital, risk_pct, stop_distance, price):
    """
    基于风险的仓位计算
    risk_pct: 单笔风险占总资金比例 (5%)
    stop_distance: 止损距离 (价格 - 止损价)
    """
    risk_amount = capital * risk_pct  # 最大亏损金额
    position_size = risk_amount / stop_distance  # 可买数量
    return position_size
```

def execute_entry(signal, position_size):
    # 市价单进场
    order = place_market_order(

    )

    return order

## 五、出场逻辑（核心）

### 分批止盈
```python
class BatchExit:
    """分批止盈管理"""
    
    def __init__(self, entry_price, stop_loss, position_size):
        self.entry_price = entry_price
        self.stop_loss = stop_loss
        self.risk = entry_price - stop_loss  # 单笔风险
        self.position_size = position_size
        self.remaining = position_size
        
        # 分批止盈目标
        self.targets = [
            {'ratio': 0.4, 'price': entry_price + self.risk * 2},  # 1:2
            {'ratio': 0.3, 'price': entry_price + self.risk * 3},  # 1:3
            {'ratio': 0.3, 'price': None},  # 移动止盈
        ]
        self.batch_idx = 0
    
    def check_exit(self, current_price, highest_price):
        """检查是否触发分批止盈"""
        exits = []
        
        if self.batch_idx < 2:  # 前两批固定目标
            target = self.targets[self.batch_idx]
            if current_price >= target['price']:
                exit_qty = self.position_size * target['ratio']
                exits.append({'price': target['price'], 'qty': exit_qty})
                self.remaining -= exit_qty
                self.batch_idx += 1
        
        # 第三批移动止盈
        if self.batch_idx == 2 and self.remaining > 0:
            trailing_stop = highest_price - ATR * 2.0
            if current_price <= trailing_stop:
                exits.append({'price': current_price, 'qty': self.remaining})
                self.remaining = 0
        
        return exits
```

### 移动止盈
```python
class TrailingStop:
    """移动止盈"""
    
    def __init__(self, entry_price, atr, trailing_multiplier=2.0):
        self.entry_price = entry_price
        self.atr = atr
        self.multiplier = trailing_multiplier
        self.highest_price = entry_price
        self.trailing_line = entry_price - atr * trailing_multiplier
        self.activated = False  # 是否已激活
    
    def update(self, current_price, current_atr):
        """更新移动止损线"""
        self.highest_price = max(self.highest_price, current_price)
        
        # 盈利超过 1:2 后激活
        profit_ratio = (current_price - self.entry_price) / (self.entry_price - self.stop_loss)
        if profit_ratio >= 2.0:
            self.activated = True
        
        if self.activated:
            new_line = self.highest_price - current_atr * self.multiplier
            self.trailing_line = max(self.trailing_line, new_line)
        
        return self.trailing_line
    
    def is_triggered(self, current_price):
        """检查是否触发移动止损"""
        return current_price <= self.trailing_line
```

## 六、风控模块

### 风控检查
```python
class RiskManager:
    def __init__(self, capital, config):
        self.capital = capital
        self.max_position_pct = config['max_position_pct']  # 5%
        self.max_positions = config['max_positions']  # 3
        self.daily_loss_limit = config['daily_loss_limit']  # 10%
        self.current_positions = {}
        self.daily_pnl = 0
    
    def check_entry(self, signal, position_size):
        """检查是否允许进场"""
        # 1. 仓位上限检查
        position_value = position_size * signal.price
        if position_value > self.capital * self.max_position_pct:
            return False, '仓位超限'
        
        # 2. 持仓数量检查
        if len(self.current_positions) >= self.max_positions:
            return False, '持仓数已达上限'
        
        # 3. 日止损检查
        if self.daily_pnl < -self.capital * self.daily_loss_limit:
            return False, '日止损已触发'
        
        # 4. 互斥检查
        if signal.symbol in self.current_positions:
            return False, '该品种已有持仓'
        
        return True, '通过'
```

## 七、异常处理

| 场景 | 处理策略 |
|------|----------|
| API 超时 | 重试 3 次，失败报警 |
| 部分成交 | 检查剩余仓位，调整止盈目标 |
| 交易所维护 | 暂停交易，保持现有止损 |
| 网络断开 | 本地记录状态，恢复后同步 |
| 价格跳空 | 硬止损可能滑点，记录实际成交价 |

## 八、裸仓保护

> S002 是单边策略，不存在对冲腿，但仍需保护

- [ ] 进场后立即设置止损单
- [ ] 止损单失败 → 立即市价平仓
- [ ] 异常恢复：重启后检查持仓 + 止损单一致性
- [ ] 逐路径验证：网络/重启/延迟/部分成交
