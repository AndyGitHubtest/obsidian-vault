# 主循环伪代码
for bar in bars:
    # 1. 更新数据
    update_data(bar)
    
    # 2. 计算因子
    factors = calc_factors(data)
    
    # 3. 生成信号
    signals = generate_signals(factors)
    
    # 4. 风控过滤
    valid_signals = risk_check(signals, positions)
    
    # 5. 执行交易
    for sig in valid_signals:
